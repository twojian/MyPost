## 逆置

数组反转

```c
void Reverse_List(int A[], int L, int R) {
    int m = (L + R) / 2;
    int t;
    for (int i = 0; i <= m - L; i++) {
        t = A[L + i];
        A[L + i] = A[R - i];
        A[R - i] = t;
    }
}
```

链表反转

```c
void Reverse(Sqlist *L) {
    int temp;
    for (int i = 0; i < L->length / 2; i++) {
        temp = L->data[i];
        L->data[i] = L->data[L->length - 1 - i]; // 对称交换
        L->data[L->length - 1 - i] = temp;
    }
}
```

## 删除

删除值为X的元素

```c
void Delete_DataX(int A[], int x, int n) {
    int t = 0;
    for (int i = 0; i < n; i++) {
        if (A[i] == x)
            t++;
        else
            A[i - t] = A[i];
    }
}
```

删除第k个元素并返回其值

```c
bool Delete(Sqlist *L, int k, int *e) {
    if (k < 1 || k > L->length) // 位置合法性检查
        return false;
    *e = L->data[k - 1]; // 保存被删元素
    for (int i = k; i < L->length; i++) { // 从k位置后移
        L->data[i - 1] = L->data[i];
    }
    L->length--;
    return true;
}
```

删除第一个为X

```c
bool Delete_First(Sqlist *L, int x) {
    int pos = -1;
    // 找x的位置
    for (int i = 0; i < L->length; i++) {
        if (L->data[i] == x) {
            pos = i;
            break;
        }
    }
    if (pos == -1)
        return false; // 未找到x
    // 元素前移
    for (int i = pos; i < L->length - 1; i++) {
        L->data[i] = L->data[i + 1];
    }
    L->length--;
    return true;
}
```

## 插入

有序顺序表插入x保持递增

```c
void Insert_Sq(Sqlist *L, int x) {
    int i;
    // 找到插入位置i
    for (i = 0; i < L->length; i++) {
        if (L->data[i] >= x)
            break;
    }
    // 元素后移
    for (int j = L->length; j > i; j--) {
        L->data[j] = L->data[j - 1];
    }
    L->data[i] = x;
    L->length++;
}
```

最后一个元素覆盖最小值

```c
int Search_Min(Sqlist *L) {
    if (L->length == 0)
        return -1;
    int min = L->data[0];
    int pos = 0;
    // 找最小值位置
    for (int i = 1; i < L->length; i++) {
        if (L->data[i] < min) {
            min = L->data[i];
            pos = i;
        }
    }
    // 若最小值不是最后一个元素，用最后一个覆盖
    if (pos != L->length - 1) {
        L->data[pos] = L->data[L->length - 1];
    }
    return min;
}
```
