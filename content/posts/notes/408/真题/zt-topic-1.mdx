4. ★★【2014年统考真题】系统中有多个生产者进程和多个消费者进程，共享一个能存放1000件产品的环形缓冲区（初始为空）。缓冲区未满时，生产者进程可以放入其生产的一件产品，否则等待；缓冲区未空时，消费者进程可以从缓冲区取走一件产品，否则等待。要求一个消费者进程从缓冲区连续取出10件产品后，其他消费者进程才可以取产品。请使用信号`P`、`V`（`wait()`、`signal()`）操作实现进程间的互斥与同步，要求写出完整的过程，并说明所用信号量的含义和初值。

```c
// 共享信号量（初值如上）
semaphore empty = 1000;
semaphore full  = 0;
semaphore mutex = 1;
semaphore c_lock = 1;

// 环形缓冲区与指针
item buffer[1000];
int in = 0, out = 0;

/*************** 生产者进程 ****************/
producer() {
    while (true) {
        item x = produce();        // 生产一个产品（本地操作）

        P(empty);                  // 需要一个空位
        P(mutex);                  // 进入缓冲区临界区
            buffer[in] = x;
            in = (in + 1) % 1000;
        V(mutex);                  // 离开临界区
        V(full);                   // 增加可用产品数
    }
}

/*************** 消费者进程 ****************/
consumer() {
    while (true) {
        // 申请批量锁，获得“本轮取10件”的资格
        P(c_lock);

        // 连续取满10件
        for (int i = 0; i < 10; ++i) {
            P(full);               // 等待至少有1件产品
            P(mutex);              // 进入缓冲区临界区
                item x = buffer[out];
                out = (out + 1) % 1000;
            V(mutex);              // 离开临界区
            V(empty);              // 归还一个空位

            consume(x);            // 消费该产品（本地操作）
        }

        // 释放批量锁，允许下一个消费者开始它的“10件批次”
        V(c_lock);
    }
}
```

6. ★★【2016年统考真题】某个进程调度程序采用基于优先数（priority）的调度策略，即选择优先数最小的进程运行。进程创建时由用户指定一个 `nice` 作为静态优先数。为了动态调整优先数，引入运行时间 cpuTime 和等待时间 waitTime，初值均为 0。进程处于执行态时，cpuTime 定时加 1，且 waitTime 置 0；进程处于就绪态时，cpuTime 置 0，waitTime 定时加 1。请回答下列问题：

   (1) 若调度程序只将 nice 的值作为进程的优先数，即 priority = nice，则可能会出现饥饿现象。为什么？

   nice 作为静态优先数，当不断有优先数较小的进程加入就绪队列时，nice 值较大的进程会迟迟得不到 CPU 资源造成饥饿现象。





   (2) 使用 nice、cpuTime 和 waitTime 设计一种动态优先数计算方法，以避免产生饥饿现象，并说明 waitTime 的作用。


```c
   priority = nice + cpuTime - waitTime;

```

   waitTime的作用在于调节就绪进程的等待时间，根据等待时间适当提高优先级。

8. ★★★【2019年统考真题】有 n（n≥3）名哲学家围坐在一张圆桌边，每名哲学家交替地就餐和思考。在圆桌中心有 m（m≥1）个碗，每两名哲学家之间有一根筷子。每名哲学家必须取到一个碗和两侧的筷子后，才能就餐，进餐完毕，将碗和筷子放回原位，并继续思考。为使尽可能多的哲学家同时就餐，且防止出现死锁现象，请使用信号量的 P、V 操作［`wait()`、`signal()`］描述上述过程中的互斥与同步，并说明所用信号量及初值的含义。

// 哲学家 i (1 ≤ i ≤ n)
philosophy(i){

Python 模拟

```c++
import threading
import time
import random

# --- 模拟参数 ---
N_PHILOSOPHERS = 5  # n: 哲学家数量
M_BOWLS = 3         # m: 碗的数量
SIMULATION_DURATION = 10 # 模拟持续时间（秒）

# --- 辅助函数 ---
def think(philosopher_id):
    """模拟思考"""
    think_time = random.uniform(0.01, 0.1)
    # print(f"哲学家 {philosopher_id} 正在思考...")
    time.sleep(think_time)

def eat(philosopher_id, meal_counter):
    """模拟就餐并计数"""
    eat_time = random.uniform(0.01, 0.1)
    # print(f"哲学家 {philosopher_id} 正在就餐。")
    time.sleep(eat_time)
    # 使用锁来安全地增加全局计数器
    with meal_counter[1]:
        meal_counter[0] += 1

# --- 算法1: 全局互斥锁 ---
def philosopher_mutex(p_id, chopsticks, bowl, mutex, meal_counter, stop_event):
    while not stop_event.is_set():
        think(p_id)

        mutex.acquire()
        bowl.acquire()
        chopsticks[p_id].acquire()            # 左手筷子
        chopsticks[(p_id + 1) % N_PHILOSOPHERS].acquire() # 右手筷子
        mutex.release() # 获取完资源后应立即释放全局锁，让其他人可以吃饭

        eat(p_id, meal_counter)

        chopsticks[(p_id + 1) % N_PHILOSOPHERS].release()
        chopsticks[p_id].release()
        bowl.release()

# --- 算法2: 房间法 ---
def philosopher_room(p_id, chopsticks, bowl, room, meal_counter, stop_event):
    while not stop_event.is_set():
        think(p_id)

        room.acquire()
        bowl.acquire()
        chopsticks[p_id].acquire()
        chopsticks[(p_id + 1) % N_PHILOSOPHERS].acquire()
        room.release() # 获取成功，立刻离开房间，让别人能进来尝试

        eat(p_id, meal_counter)

        chopsticks[(p_id + 1) % N_PHILOSOPHERS].release()
        chopsticks[p_id].release()
        bowl.release()

# --- 算法3: 奇偶交错法 ---
def philosopher_odd_even(p_id, chopsticks, bowl, meal_counter, stop_event):
    while not stop_event.is_set():
        think(p_id)

        bowl.acquire()
        if p_id % 2 == 0: # 偶数号: 先左后右
            chopsticks[p_id].acquire()
            chopsticks[(p_id + 1) % N_PHILOSOPHERS].acquire()
        else: # 奇数号: 先右后左
            chopsticks[(p_id + 1) % N_PHILOSOPHERS].acquire()
            chopsticks[p_id].acquire()

        eat(p_id, meal_counter)

        chopsticks[p_id].release()
        chopsticks[(p_id + 1) % N_PHILOSOPHERS].release()
        bowl.release()

def run_simulation(philosopher_logic, n, m, description):
    """通用模拟运行器"""
    chopsticks = [threading.Semaphore(1) for _ in range(n)]
    bowl = threading.Semaphore(m)
    meal_counter = [0, threading.Lock()] # [count, lock]
    stop_event = threading.Event()

    # 根据不同算法设置额外信号量
    extra_semaphores = {}
    if philosopher_logic == philosopher_mutex:
        extra_semaphores['mutex'] = threading.Semaphore(1)
    elif philosopher_logic == philosopher_room:
        # 房间容量不能超过哲学家总数-1
        room_size = min(n - 1, m * 2) # 一个更优化的想法，房间容量也受碗数限制
        extra_semaphores['room'] = threading.Semaphore(room_size)

    threads = []
    for i in range(n):
        thread = threading.Thread(target=philosopher_logic, args=(i, chopsticks, bowl, *extra_semaphores.values(), meal_counter, stop_event))
        threads.append(thread)
        thread.start()

    print(f"\n--- 开始模拟: {description} ---")
    print(f"哲学家数量: {n}, 碗数量: {m}")
    time.sleep(SIMULATION_DURATION)

    stop_event.set()
    for thread in threads:
        thread.join()

    print(f"--- 模拟结束 ---")
    print(f"在 {SIMULATION_DURATION} 秒内, 总共就餐次数: {meal_counter[0]}")
    return meal_counter[0]

if __name__ == "__main__":
    print("="*40)
    print("开始并发性测试...")
    print("="*40)

    results = {}

    # 运行算法1
    results['全局互斥锁'] = run_simulation(philosopher_mutex, N_PHILOSOPHERS, M_BOWLS, "算法1: 全局互斥锁")

    # 运行算法2
    results['房间法'] = run_simulation(philosopher_room, N_PHILOSOPHERS, M_BOWLS, "算法2: 房间法")

    # 运行算法3
    results['奇偶交错法'] = run_simulation(philosopher_odd_even, N_PHILOSOPHERS, M_BOWLS, "算法3: 奇偶交错法")

    print("\n\n" + "="*40)
    print("           最终结果对比")
    print("="*40)
    for name, count in results.items():
        print(f"{name}:\t{count} 次就餐")
    print("="*40)
```
