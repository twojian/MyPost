## 一、树（Tree）通用操作函数（核心模板）

### 1️⃣ 计算二叉树高度（利用“递归闭包性质”）

```c
int height(TreeNode* root) {
    if (root == NULL) return 0;
    int lh = height(root->left);
    int rh = height(root->right);
    return (lh > rh ? lh : rh) + 1;
}


```

**决定性性质**:

树无环 + 高度定义天然递归

👉 不需要 visited，不需要全局变量

---

### 2️⃣ 判断是否为叶子结点（结构性判定）

```c
bool isLeaf(TreeNode* node) {
    return node && node->left == NULL && node->right == NULL;
}


```

**决定性性质**:

叶子结点只由“度 = 0”决定，与值、层次无关

---

### 3️⃣ 中序遍历判定 BST（利用“中序严格有序”）

```c
bool isValidBST(TreeNode* root, long* prev) {
    if (root == NULL) return true;

    if (!isValidBST(root->left, prev))
        return false;

    if (root->val <= *prev)
        return false;

    *prev = root->val;

    return isValidBST(root->right, prev);
}


```

调用方式:

```c
long prev = LONG_MIN;
bool ok = isValidBST(root, &prev);


```

**决定性性质**:

BST 的全局有序性只能通过中序体现

---

### 5️⃣ 判断 AVL 树（一次递归完成）

```c
int checkAVL(TreeNode* root) {
    if (root == NULL) return 0;

    int lh = checkAVL(root->left);
    if (lh == -1) return -1;

    int rh = checkAVL(root->right);
    if (rh == -1) return -1;

    if (abs(lh - rh) > 1)
        return -1;

    return (lh > rh ? lh : rh) + 1;
}


```

调用:

```c
bool isAVL = (checkAVL(root) != -1);


```

**决定性性质**:

AVL 平衡条件必须对“任意结点”成立

👉 自底向上，发现一次失衡立即终止

---

### 6️⃣ 利用前序 + 中序构造二叉树

```c
TreeNode* build(int* pre, int pl, int pr,
                int* in,  int il, int ir) {
    if (pl > pr) return NULL;

    TreeNode* root = newNode(pre[pl]);

    int k = il;
    while (in[k] != pre[pl]) k++;

    int leftSize = k - il;

    root->left = build(pre, pl+1, pl+leftSize,
                       in, il, k-1);
    root->right = build(pre, pl+leftSize+1, pr,
                        in, k+1, ir);
    return root;
}


```

**决定性性质**:

前序确定根，中序确定左右边界

```c
int countNodes(TreeNode* root) {
```

### 3️⃣ 判断是否为完全二叉树（层序判定）

```c
bool isComplete(TreeNode* root) {
    queue<TreeNode*> q;
    q.push(root);
    bool mustBeLeaf = false;

    while (!q.empty()) {
        TreeNode* cur = q.front(); q.pop();
        if (cur == NULL) {
            mustBeLeaf = true;
        } else {
            if (mustBeLeaf) return false;
            q.push(cur->left);
            q.push(cur->right);
        }
    }
    return true;
}


```

**决定性性质**

完全二叉树的“空缺只能出现在层序遍历末尾”

### 4️⃣ 判断两棵树是否结构相同

```c
bool sameTree(TreeNode* a, TreeNode* b) {

    if (a == NULL || b == NULL) return false;

}


```

**决定性性质**

### 5️⃣ 判断是否为对称二叉树（镜像）

```c
bool isMirror(TreeNode* a, TreeNode* b) {
    if (a == NULL && b == NULL) return true;
    if (a == NULL || b == NULL) return false;
    return isMirror(a->left, b->right) &&
           isMirror(a->right, b->left);
}


```

**决定性性质**

对称性是“结构镜像”，与值是否相同无关（若题目未要求）

---

### 6️⃣ 最近公共祖先（LCA，普通二叉树）链式存储

```c
TreeNode* LCA(TreeNode* root, TreeNode* p, TreeNode* q) {
    if (root == NULL || root == p || root == q)
        return root;

    TreeNode* left = LCA(root->left, p, q);
    TreeNode* right = LCA(root->right, p, q);

    if (left && right) return root;
    return left ? left : right;
    //如果在某一侧找到了 p 或 q（或已经确定的 LCA），
		//就把这个结果原封不动地向上传递；
		//如果两边都没找到，就返回 NULL。
}


```

**决定性性质**

树中路径唯一 ⇒ 最近公共祖先唯一

---

### 7️⃣ BST 中的 LCA（利用有序性）

```c
TreeNode* bstLCA(TreeNode* root, int p, int q) {
    if (root == NULL) return NULL;
    if (p < root->val && q < root->val)
        return bstLCA(root->left, p, q);
    if (p > root->val && q > root->val)
        return bstLCA(root->right, p, q);
    return root;
}


```

**决定性性质**

BST 的全序关系直接决定搜索方向

### 8️⃣ 判断是否为退化树（链）

```c
bool isDegenerate(TreeNode* root) {
    if (root == NULL) return true;
    if (root->left && root->right) return false;
    return isDegenerate(root->left ? root->left : root->right);
}


```

**决定性性质**

### 9️⃣ 将二叉树转换为双向链表（中序）

```c
void treeToList(TreeNode* root, TreeNode** prev) {
    if (root == NULL) return;
    treeToList(root->left, prev);
    root->left = *prev;
    if (*prev) (*prev)->right = root;
    *prev = root;
    treeToList(root->right, prev);
}


```

**决定性性质**

中序遍历天然给出“线性顺序”

### 0️⃣ 验证 n0 = n2 + 1（结构校验）

```c
void countDegree(TreeNode* root, int* n0, int* n2) {
    if (root == NULL) return;
    if (root->left == NULL && root->right == NULL)
        (*n0)++;
    else if (root->left && root->right)
        (*n2)++;
    countDegree(root->left, n0, n2);
    countDegree(root->right, n0, n2);
}


```

**决定性性质**

所有二叉树必须满足 n₀ = n₂ + 1

<details><summary>非递归操作</summary>

```c
void visit(BTNode *T) {
    printf("%d ", T->data);
}

void traverse(BTNode *T, int type) {
    // type: 0=后序, 1=中序, 2=先序
    if (T == NULL)
        return;
    BTNode *stk[MaxSize];  // 主栈:存节点
    int tags[MaxSize];     // 标记栈:0=节点，1=子树
    int top = -1;

    stk[++top] = T;
    tags[top] = 1;

    while (top != -1) {
        BTNode *node = stk[top];
        int tag = tags[top--];

        if (tag == 0) {
            visit(node);
        } else {
            // 按“逆序”入栈（保证出栈顺序正确）
            if (type == 0) {  // 后序:节点→右子树→左子树
                stk[++top] = node;
                tags[top] = 0;
                if (node->rchild != NULL) {
                    stk[++top] = node->rchild;
                    tags[top] = 1;
                }
                if (node->lchild != NULL) {
                    stk[++top] = node->lchild;
                    tags[top] = 1;
                }
            } else if (type == 1) {  // 中序:右子树→节点→左子树
                if (node->rchild != NULL) {
                    stk[++top] = node->rchild;
                    tags[top] = 1;
                }
                stk[++top] = node;
                tags[top] = 0;
                if (node->lchild != NULL) {
                    stk[++top] = node->lchild;
                    tags[top] = 1;
                }
            } else if (type == 2) {  // 先序:右子树→左子树→节点
                if (node->rchild != NULL) {
                    stk[++top] = node->rchild;
                    tags[top] = 1;
                }
                if (node->lchild != NULL) {
                    stk[++top] = node->lchild;
                    tags[top] = 1;
                }
                stk[++top] = node;
                tags[top] = 0;
            }
        }
    }
}
```

</details>

<details><summary>先序+中序</summary>

```c
BiTNode* PreInCreate(ElemType pre[], ElemType in[], int preL, int preR, int inL, int inR) {
    if (preL > preR) return NULL; // 避免越界
    BiTNode* root = (BiTNode*)malloc(sizeof(BiTNode));
    root->data = pre[preL]; // 先序遍历的第一个节点为当前根节点

    int i = inL;
    while (i <= inR && in[i] != root->data) i++; // 定位根节点在中序遍历中的位置

    int leftLen = i - inL; // 左子树节点数目
    root->lchild = PreInCreate(pre, in, preL + 1, preL + leftLen, inL, i - 1);
    root->rchild = PreInCreate(pre, in, preL + leftLen + 1, preR, i + 1, inR);
    return root;
}
```

```c
BiTNode* InPostCreate(ElemType in[], ElemType post[], int inL, int inR, int postL, int postR) {
    if (postL > postR) return NULL; // 避免越界
    BiTNode* root = (BiTNode*)malloc(sizeof(BiTNode));
    root->data = post[postR]; // 后序遍历的最后一个节点为当前根节点

    int i = inL;

    int leftLen = i - inL; // 左子树节点数目
    root->lchild = InPostCreate(in, post, inL, i - 1, postL, postL + leftLen - 1);
    root->rchild = InPostCreate(in, post, i + 1, inR, postL + leftLen, postR - 1);
    return root;
}
```

<details><summary>判断平衡二叉树</summary>

```c
bool balance = true;
bool isBalanced(TreeNode *root) {
    helper(root);
    return balance;
}

int helper(TreeNode *root) {
    if (root == NULL)
        return 0;
    int left = helper(root->lchild);
    int right = helper(root->rchild);
    // 高度差超过1则不平衡
    if (right - left > 1 || left - right > 1)
        balance = false;
    return max(left, right) + 1;
}
```

```c
bool IsBST(BiTree T, int min_val, int max_val) {
    if (T == NULL)
        return true;
    // 当前节点值需在[min_val, max_val]范围内
    if (T->data < min_val || T->data > max_val)
        return false;
    // 左子树值范围:[min_val, T->data)
    // 右子树值范围:(T->data, max_val]
    return IsBST(T->lchild, min_val, T->data) 
        && IsBST(T->rchild, T->data, max_val);
}
```

```c
    SqQueue Q;
    InitQueue(Q);
    BiTNode *temp = T;

        DeQueue(Q, temp);

            }
        }
```

# 并查集

<details><summary>查找连通分量个数</summary>

```c
#define SIZE 100
int UFSets[SIZE]; // 并查集数组:根节点存集合大小（负数），子节点存父节点

// 初始化并查集:每个顶点自成一个集合【考研必背】
void Initial(int S[]) {
    for (int i = 0; i < SIZE; i++) {
        S[i] = -1; // 初始根节点，值为-1代表集合大小为1
    }
}

// 查找x的根节点（带路径压缩）【考研核心优化:减少后续查找时间】
int Find(int S[], int x) {
    while (S[x] >= 0) { // 不是根节点则向上找
        x = S[x];
    }
    return x; // 返回根节点
}

// 合并x和y所在的集合【考研易错点:按根节点合并，不是直接合并x和y】
void Union(int S[], int x, int y) {
    int Root1 = Find(S, x); // 找x的根
    int Root2 = Find(S, y); // 找y的根
    if (Root1 != Root2) {   // 不在同一集合才合并
        S[Root2] = Root1;   // 合并集合（简单合并，可优化为按秩合并）
    }
}

// 统计图的连通分量个数【考研常考应用】
int CountComponents(MGraph G) {
    Initial(UFSets);
    int n = G.vernum;
    // 遍历所有边，合并连通顶点
    for (int u = 0; u < n; u++) {
        for (int v = u + 1; v < n; v++) {
            if (G.edge[u][v] != INF) {
                Union(UFSets, u, v);
            }
        }
    }
    int count = 0;
    // 统计根节点个数:根节点值为负数【考研判断依据】
    for (int i = 0; i < n; i++) {
        if (UFSets[i] < 0) {
            count++;
        }
    }
    return count;
}
```

```c
int find(int arr[], int i) {
    int root = i;
    // 找到根节点
    while (arr[root] != -1)
        root = arr[root];
    // 路径压缩:将路径上节点直接指向根
    while (i != root) {
        int temp = arr[i];
        arr[i] = root;
        i = temp;
    }
    return root;
}
```

```c
void union(int arr[], int r1, int r2) {
    if (arr[r1] == arr[r2])
        return 1; // 已在同一集合
    // 大集合合并小集合
    if (arr[r1] > arr[r2]) {
        arr[r1] += arr[r2];
        arr[r2] = r1;
    } else {
        arr[r2] += arr[r1];
        arr[r1] = r2;
    }
}
```
