### 43️⃣ 统计强连通分量个数（Tarjan 核心）

```c
void tarjan(int u) {
    dfn[u] = low[u] = ++time;
    stk.push(u);
    inStack[u] = true;

    for (int v : adj[u]) {
        if (!dfn[v]) {
            tarjan(v);
            low[u] = min(low[u], low[v]);
        } else if (inStack[v]) {
            low[u] = min(low[u], dfn[v]);
        }
    }

    if (dfn[u] == low[u]) {
        while (true) {
            int x = stk.top(); stk.pop();
            inStack[x] = false;
            if (x == u) break;
        }
        sccCnt++;
    }
}


```

## 最小生成树

<details><summary>Prim</summary>

```c
// Prim算法:从起点s出发构造最小生成树，适合稠密图【考研高频考点】
int Prim(MGraph G, int s) {
    int n = G.vernum;
    bool visited[MAXV] = {false}; // 标记顶点是否加入生成树
    int dist[MAXV]; // dist[v]:v到生成树的最短距离【核心数组】
    int ans = 0;    // 最小生成树的总权值

    // 初始化距离数组:起点s到各顶点的直接边权
    for (int i = 0; i < n; i++) {
        dist[i] = G.edge[s][i];
    }
    visited[s] = true; // 起点加入生成树

    // 循环n-1次:生成树有n个顶点，需选n-1条边【考研关键】
    for (int i = 1; i < n; i++) { 
        int u = -1, minDist = INF;
        // 第一步:找未加入生成树的、距离最小的顶点u【考研核心步骤】
        for (int j = 0; j < n; j++) {
            if (!visited[j] && dist[j] < minDist) {
                minDist = dist[j];
                u = j;
            }
        }
        // 【考研判断:u=-1说明图不连通，无法生成最小生成树】
        if (u == -1) return -1; 
        visited[u] = true; // u加入生成树
        ans += minDist;    // 累加边权
        // 第二步:更新未加入顶点到生成树的距离【考研核心步骤】
        for (int v = 0; v < n; v++) {
            // 条件:v未加入 且 边u-v的权值小于当前dist[v]
            if (!visited[v] && G.edge[u][v] < dist[v]) {
                dist[v] = G.edge[u][v];
            }
        }
    }
    return ans; // 返回最小生成树总权值
}
```

<details><summary>哈密顿路径</summary>

```c++
/**
 * @param G: 邻接矩阵，表示无向图，按二维数组的方式用下标即可访问内部元素
 * @param n: 图中顶点的数量
 * @return: 返回布尔值，表示是否存在从0号顶点出发的哈密顿路径
 */

bool dfs(int** G, vector<char>& visited, int k, int visitedCount) {
    // 如果已访问所有顶点，说明找到了哈密顿路径
    if (visitedCount == visited.size()) return true;

    // 遍历所有顶点
    for (int i = 0; i < visited.size(); i++) {
        // 如果顶点i未被访问，且顶点k到顶点i有边
        if (!visited[i] && G[k][i]) {
            visited[i] = 1;  // 标记顶点i为已访问
            // 递归搜索，如果找到哈密顿路径则返回true
            if (dfs(G, visited, i, visitedCount + 1)) return true;
            visited[i] = 0;  // 回溯，撤销访问标记
        }
    }
    return false;  // 未找到哈密顿路径
}

bool hasHamiltonianPath(int** G, int n) {
    vector<char>; visited(n, 0);  // 初始化访问标记数组，0表示未访问
    visited[0] = 1;  // 标记起始顶点（0号顶点）为已访问
    dfs(G, visited, 0, 1);  // 从0号顶点开始深度优先搜索

    // 检查是否所有顶点都被访问过
    for (int i = 0; i < n; i++) {
        if (visited[i] == 0) return false;  // 存在未访问的顶点，无哈密顿路径
    }

    return true;  // 所有顶点都被访问过，存在哈密顿路径
}
```

## 拓扑排序

<details><summary>Khan BFS</summary>

```c++
/**
 * @param graph: 邻接表表示的有向图
 * @param n: 图中顶点的数量
 * @return: 返回拓扑排序结果，如果不存在则返回空vector
 */
vector<int> topologicalSort(vector<vector<int>>& graph, int n) {
    // 计算每个节点的入度
    vector<int> indegree(n, 0);
    for (int i = 0; i < n; i++) {
        for (int neighbor : graph[i]) {
            indegree[neighbor]++;
        }
    }
    
    // 将所有入度为0的节点加入队列
    queue<int> q;
    for (int i = 0; i < n; i++) {
        if (indegree[i] == 0) {
            q.push(i);
        }
    }
    
    vector<int> result;
    while (!q.empty()) {
        int curr = q.front();
        q.pop();
        result.push_back(curr);
        
        // 将当前节点指向的所有邻居节点的入度减1
        for (int neighbor : graph[curr]) {
            // 如果入度变为0，加入队列
            if (!--indegree[neighbor]) q.push(neighbor);
        }
    }
    
    // 检查是否存在环
    if (result.size() != n) {
        return {}; // 存在环，无法进行拓扑排序
    }
    
    return result;
}
```

## 最短路径

<details><summary>Floyd算法</summary>

```c
void Floyd(MGraph G) {  // O(n³)
    int i, j, k;  // k为中转顶点
    int A[MaxSize][MaxSize];
    int path[MaxSize][MaxSize];
    int n = G.vernum;
    // 初始化A和path矩阵
    for (i = 0; i < n; i++) {
        for (j = 0; j < n; j++) {
            if (G.edge[i][j] == 0 && i != j) {  // 假设0表示没有边，非对角元素
                A[i][j] = INF;  // 无边设置为无穷大
            } else {
                A[i][j] = G.edge[i][j];
            }
            path[i][j] = -1;  // -1表示没有中转点
        }
    }

    // 执行Floyd-Warshall算法
    for (k = 0; k < n; k++) {
        for (i = 0; i < G.vernum; i++) {
            for (j = 0; j < n; j++) {
                // 如果通过k能得到更短的路径
                if (A[i][k] != INF && A[k][j] != INF && A[i][k] + A[k][j] < A[i][j]) {
                    A[i][j] = A[i][k] + A[k][j];
                    path[i][j] = k;  // 更新路径，k为中转点
                }
            }
        }
    }

    // 打印最短路径矩阵A
    for (i = 0; i < G.vernum; i++) {
        for (j = 0; j < G.vernum; j++) {
            if (A[i][j] == INF) {
                printf("INF ");
            } else {
                printf("%d ", A[i][j]);
            }
        }
        printf("\n");
    }
}
```

```c
// Dijkstra算法:求起点s到所有顶点的最短路径，不能处理负权边【考研高频考点】
void Dijkstra(MGraph G, int s) {
    int n = G.vernum;
    bool visited[MAXV] = {false}; // 标记顶点是否已确定最短路径
    int dist[MAXV]; // dist[v]:s到v的当前最短路径长度【核心数组】
		int path[MAXV];
    // 初始化距离数组:起点s到各顶点的直接边权
    for (int i = 0; i < n; i++) {
        dist[i] = G.edge[s][i];
        path[i]= -1;
    }
    dist[s] = 0;       // 起点到自身距离为0【考研易错点:必须单独写】
    visited[s] = true; // 起点最短路径已确定

    // 循环n-1次:确定n-1个顶点的最短路径
    for (int i = 1; i < n; i++) {
        int u = -1, minDist = INF;
        // 第一步:找未确定最短路径的、距离最小的顶点u【考研核心】
        for (int j = 0; j < n; j++) {
            if (!visited[j] && dist[j] < minDist) {
                minDist = dist[j];
                u = j;
            }
        }
        // 图不连通，提前返回
        if (u == -1) return; 
        visited[u] = true; // u的最短路径已确定
        // 第二步:通过u更新其他顶点的距离【考研核心】
        for (int v = 0; v < n; v++) {
            // 条件:v未确定 且 存在边u-v 且 经过u的路径更短
            if (!visited[v] && G.edge[u][v] != INF && dist[v] > dist[u] + G.edge[u][v]) {
                dist[v] = dist[u] + G.edge[u][v];
                path[v]=u;
            }
        }
    }
}
```
