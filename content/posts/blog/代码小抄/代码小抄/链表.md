## 插入

<details><summary>头插法/尾插法建表</summary>

```c
typedef struct LNode {
    int data;
    struct LNode *next;
} LNode, *LinkList;

// 头插法（输入-1结束）
void Head_Insert(LinkList *L) {
    *L = (LinkList)malloc(sizeof(LNode));
    (*L)->next = NULL;
    int x;
    scanf("%d", &x);
    while (x != -1) {
        LNode *s = (LNode *)malloc(sizeof(LNode));
        s->data = x;
        s->next = (*L)->next;
        (*L)->next = s;
        scanf("%d", &x);
    }
}

// 尾插法（输入-1结束）
void Tail_Insert(LinkList *L) {
    *L = (LinkList)malloc(sizeof(LNode));
    (*L)->next = NULL;
    LNode *r = *L; // 尾指针
    int x;
    scanf("%d", &x);
    while (x != -1) {
        LNode *s = (LNode *)malloc(sizeof(LNode));
        s->data = x;
        s->next = NULL;
        r->next = s;
        r = s; // 更新尾指针
        scanf("%d", &x);
    }
}
```

```c
bool New_Insert(LinkList L, int x) {
    LNode *s = (LNode *)malloc(sizeof(LNode));
    s->data = x;
    LNode *r = L; // r指向头结点，遍历找插入位置
    while (r->next != NULL && r->next->data < x) {
        r = r->next; // 找到第一个>=x的节点的前驱
    }
    // 插入s到r之后
    s->next = r->next;
    r->next = s;
    return true;
}
```

<details><summary>移动最小值节点到链表头部</summary>

```c
void Move_toFront_min(LinkList L) {
    if (L == NULL || L->next == NULL)
        return;
    LinkList min = L->next;   // 最小值节点
    LinkList min_prev = L;    // 最小值节点的前驱
    LinkList curr = L->next;
    LinkList prev = L;

    // 找最小值节点及其前驱
    while (curr != NULL) {
        if (curr->data < min->data) {
            min = curr;
            min_prev = prev;
        }
        prev = curr;
        curr = curr->next;
    }

    // 若最小值已在头部，无需移动
    if (min_prev == L)
        return;

    // 断开min与原前驱的连接
    min_prev->next = min->next;
    // 将min插入到头部
    min->next = L->next;
    L->next = min;
}
```

```c
bool Delete_First(LinkList L, int x) {
    LNode *p = L;
    while (p->next != NULL) {
        if (p->next->data == x) {
            LNode *q = p->next;
            p->next = q->next;
            free(q);
            return true;
        }
        p = p->next;
    }
    return false; // 未找到x
}
```

<details><summary>单链表最小值节点的奇偶处理</summary>

```c
void Process(LinkList L) {
    if (L == NULL || L->next == NULL)
        return;
    LinkList min = L->next;       // 最小值节点
    LinkList min_prev = L;        // 最小值节点的前驱
    LinkList curr = L->next;
    LinkList prev = L;

    // 找最小值节点及其前驱
    while (curr != NULL) {
        if (curr->data < min->data) {
            min = curr;
            min_prev = prev;
        }
        prev = curr;
        curr = curr->next;
    }

    if (min->data % 2 == 0) {
        // 偶数:删除其后继节点
        if (min->next != NULL) {
            LinkList temp = min->next;
            min->next = temp->next;
            free(temp);
        }
    } else {
        // 奇数:与后继节点交换（节点交换，非数值）
        if (min->next != NULL) {
            LinkList next_node = min->next;
            // 调整前驱节点的指向
            min_prev->next = next_node;
            // 交换min和next_node的后继
            min->next = next_node->next;
            next_node->next = min;
        }
    }
}
```

## 查找

<details><summary>单链表的中心对称判断</summary>

```c
typedef struct node {
    int data;
    struct node *next;
} node, *linklist;

bool isPalindrome(linklist L) {
    if (L == NULL || L->next == NULL)
        return true;
    linklist slow = L->next, fast = L->next;
    // 快慢指针找中点
    while (fast->next != NULL && fast->next->next != NULL) {
        slow = slow->next;
        fast = fast->next->next;
    }

    // 逆置后半段
    linklist mid = slow->next;
    linklist prev = NULL, curr = mid;
    while (curr != NULL) {//头插法
        linklist next = curr->next;//中间节点存
        curr->next = prev;//指向
        prev = curr;
        curr = next;
    }

    // 比较前半段与逆置后的后半段
    linklist p1 = L->next, p2 = prev;
    bool res = true;
    while (p2 != NULL) {
        if (p1->data != p2->data) {
            res = false;
            break;
        }
        p1 = p1->next;
        p2 = p2->next;
    }

    // 恢复后半段（可选）
    curr = prev;
    prev = NULL;
    while (curr != NULL) {
        linklist next = curr->next;
        curr->next = prev;
        prev = curr;
        curr = next;
    }
    slow->next = prev;

    return res;
}
```

## 逆置

<details><summary>单链表原地逆置（空间O(1)）</summary>

```c
void Reverse(LinkList L) {
    LinkList p = L->next, q;
    L->next = NULL; // 头结点置空
    while (p != NULL) {
        q = p->next; // 暂存后继
        p->next = L->next; // 新节点指向头结点的后继
        L->next = p; // 头结点指向新节点
        p = q; // 处理下一个节点
    }
}
```
